# VSBC camera + DIC scripts: the CAMERA environment (.venv-camera).
# Raspberry Pi 5 (64-bit Pi OS Bookworm or Trixie) or a PC.
#
# Install into the separate camera environment, NOT the app's .venv:
#   bash scripts/install_pi.sh --camera            # does all of this on the Pi
#   python3 -m venv ~/VSBC_driver/.venv-camera
#   ~/VSBC_driver/.venv-camera/bin/pip install -r camera/requirements.txt
#
# Why a separate environment: pyvale needs PyQt6, opencv-python (which brings
# its own Qt) and NumPy 2. The app's .venv uses the Pi's apt PyQt5, NumPy 1 and
# matplotlib. Mixed in one environment, pyqtgraph would switch the GUI to PyQt6,
# two Qt libraries would load into one process, and apt's matplotlib would stop
# working. The GUI runs everything camera-related in this environment, as
# separate processes, so the two never meet.
#
# On the laptop, the existing conda "dic" environment works too: set
#   [camera] python = "C:/Users/<you>/miniforge3/envs/dic/python.exe"
# in config.toml.

# Basler camera (the ARM64 wheel includes the pylon runtime)
pypylon

# DIC + strain: the version the scripts were developed on. It has prebuilt
# ARM64 (aarch64) packages for Python 3.11 to 3.14, so it installs on Pi OS
# Bookworm (3.11) and Trixie (3.13). Older releases are tied to one Python
# version each (2026.4.0 to 3.11 only), which is why pinning those failed.
pyvale==2026.10.2

# Images, preview windows (standalone use), crack tracker, video
opencv-python

# Usually installed with pyvale; listed so nothing is missing
numpy
scipy
matplotlib
pillow
pandas
